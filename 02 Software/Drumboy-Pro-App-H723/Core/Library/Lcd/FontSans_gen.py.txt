# Builds FontSans.h: Liberation Sans (SIL OFL 1.1), anti-aliased 4-bit glyphs, ASCII 32-126,
# one size per stock font slot, sized so CAPITAL height = the stock font's height.
from PIL import Image, ImageDraw, ImageFont
FD="/usr/share/fonts/truetype/liberation/"
SLOTS=[("05x07",7,"Regular"),("07x09",9,"Regular"),("10x14",14,"Regular"),("14x18",18,"Bold")]
out=["// Custom: Liberation Sans (SIL Open Font License 1.1, metric match for Arial), generated by",
     "// fontgen/gen.py. 4-bit anti-aliased glyphs for ASCII 32-126, one size per stock font slot,",
     "// sized so a capital letter is exactly as tall as the stock pixel font.",
     "#ifndef __FONT_SANS_H","#define __FONT_SANS_H","#include <stdint.h>","",
     "struct SansGlyph {","    uint16_t offset;   // into the font's data (bytes, 2 pixels per byte, rows padded to whole bytes)",
     "    uint8_t w, h;      // bitmap size","    int8_t xoff;       // from pen to bitmap left","    int8_t yoff;       // from baseline to bitmap top (negative = above)",
     "    uint8_t adv;       // advance in 1/4 pixels","};","",
     "struct SansFont {","    const SansGlyph* glyphs;","    const uint8_t* data;","    uint8_t capH;      // = stock font height","    uint8_t desc;      // deepest pixel below the baseline","};",""]
summary=[]
for name,cap,weight in SLOTS:
    path=FD+f"LiberationSans-{weight}.ttf"
    best=None
    for s10 in range(60,400):
        f=ImageFont.truetype(path,s10/10)
        bb=f.getbbox("H")
        if bb[3]-bb[1]==cap: best=s10/10; 
        if bb[3]-bb[1]>cap: break
    f=ImageFont.truetype(path,best)
    asc,_=f.getmetrics()
    glyphs=[]; data=[]; maxdesc=0
    for code in range(32,127):
        ch=chr(code)
        img=Image.new("L",(int(best*3)+8,int(best*3)+8),0)
        ImageDraw.Draw(img).text((4,4),ch,font=f,fill=255)
        bbox=img.getbbox()
        adv=int(round(f.getlength(ch)*4))
        if not bbox:
            glyphs.append((len(data),0,0,0,0,adv)); continue
        x0,y0,x1,y1=bbox
        w,h=x1-x0,y1-y0
        baseline=4+asc
        yoff=y0-baseline; xoff=x0-4
        maxdesc=max(maxdesc,y1-baseline)
        off=len(data)
        px=img.load()
        for yy in range(y0,y1):
            row=[(px[xx,yy]+8)//17 for xx in range(x0,x1)]
            if len(row)%2: row.append(0)
            for i in range(0,len(row),2): data.append((row[i]<<4)|row[i+1])
        glyphs.append((off,w,h,xoff,yoff,adv))
    out.append(f"// {name}: {weight}, {best} px, capital height {cap}, {len(data)} bytes")
    out.append(f"const SansGlyph kSansGlyphs_{name}[95] = {{")
    for code,g in zip(range(32,127),glyphs):
        out.append(f"    {{{g[0]}, {g[1]}, {g[2]}, {g[3]}, {g[4]}, {g[5]}}},   // {repr(chr(code))}")
    out.append("};")
    out.append(f"const uint8_t kSansData_{name}[{len(data)}] = {{")
    for i in range(0,len(data),24):
        out.append("    "+", ".join(f"0x{b:02X}" for b in data[i:i+24])+",")
    out.append("};")
    out.append(f"const SansFont kSansFont_{name} = {{kSansGlyphs_{name}, kSansData_{name}, {cap}, {maxdesc}}};")
    out.append("")
    summary.append((name,best,len(data),maxdesc))
out.append("#endif")
open("FontSans.h","w").write("\n".join(out)+"\n")
print(summary)
